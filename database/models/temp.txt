"""
Base models for QA Manager.
Simple, explicit, no magic.

from datetime import datetime, timezone
from uuid import UUID, uuid4
from sqlmodel import SQLModel, Field


def utc_now() -> datetime:
    """Get current UTC time."""
    return datetime.now(timezone.utc)


class BaseModel(SQLModel):
    """
    Base model with common fields.
    All models inherit this to get: id, created_at, updated_at, deleted_at
    """
    
    id: UUID = Field(default_factory=uuid4, primary_key=True)
    created_at: datetime = Field(default_factory=utc_now)
    updated_at: datetime = Field(default_factory=utc_now)
    deleted_at: datetime | None = Field(default=None)
    
    class Config:
        from_attributes = True


class TenantBaseModel(BaseModel):
    """Base model for tenant-scoped tables."""
    tenant_id: UUID = Field(foreign_key="tenants.id", index=True)

def ensure_utc(dt: datetime | None) -> datetime | None:
    """Ensure datetime is timezone-aware (UTC)."""
    if dt is None:
        return None
    if dt.tzinfo is None:
        return dt.replace(tzinfo=timezone.utc)
    return dt.astimezone(timezone.utc)


def now_utc() -> datetime:
    """Get current UTC datetime."""
    return datetime.now(timezone.utc)

"""
Project and TestSuite models.

Project represents a repository/codebase being tested.
TestSuite is a logical grouping of tests (file, module, feature).
"""

from uuid import UUID

from sqlmodel import Field, SQLModel, Relationship, Column, Index

from .base import TenantBaseModel


# =============================================================================
# Database Models
# =============================================================================

class Project(TenantBaseModel, table=True):
    """
    Project table - represents a repository/codebase being tested.
    """
    
    __tablename__ = "projects"
    
    # Basic info
    name: str = Field(max_length=255)
    description: str | None = Field(
        default=None,
        sa_column=Column(TEXT, nullable=True),
    )
    
    # Repository
    repository_url: str | None = Field(default=None, max_length=500)
    repository_type: str | None = Field(default="git", max_length=50)
    default_branch: str = Field(default="main", max_length=100)
    
    # Configuration (JSONB for flexibility)
    config: dict = Field(
        default_factory=dict,
        sa_column=Column(JSONB, nullable=False, server_default="{}"),
    )
    
    # Relationships
    test_suites: list["TestSuite"] = Relationship(back_populates="project")
    
    __table_args__ = (
        Index("idx_project_tenant", "tenant_id"),
        Index("idx_project_tenant_name", "tenant_id", "name"),
    )


class TestSuite(TenantBaseModel, table=True):
    """
    TestSuite table - logical grouping of test cases.
    
    Supports hierarchy (parent_id) for nested suites.
    Examples:
    - tests/unit/api (path)
    - tests/integration/database (path)
    """
    
    __tablename__ = "test_suites"
    
    # Hierarchy
    project_id: UUID = Field(foreign_key="projects.id", index=True)
    parent_id: UUID | None = Field(
        default=None,
        foreign_key="test_suites.id",
        nullable=True,
    )
    
    # Identity
    name: str = Field(max_length=500)
    description: str | None = Field(
        default=None,
        sa_column=Column(TEXT, nullable=True),
    )
    
    # Path (for file-based suites)
    path: str = Field(max_length=1000)  # e.g., "tests/unit/api"
    
    # Classification
    tags: list[str] = Field(
        default_factory=list,
        sa_column=Column(ARRAY(TEXT), nullable=False, server_default="{}"),
    )
    category: str | None = Field(default=None, max_length=100)  # 'unit', 'integration', 'e2e'
    
    # Metadata
    meta_data: dict = Field(
        default_factory=dict,
        sa_column=Column(JSONB, nullable=False, server_default="{}"),
    )
    
    # Relationships
    project: Project = Relationship(back_populates="test_suites")
    parent: "TestSuite | None" = Relationship(
        back_populates="children",
        sa_relationship_kwargs={
            "remote_side": "TestSuite.id",
            "foreign_keys": "[TestSuite.parent_id]",
        },
    )
    children: list["TestSuite"] = Relationship(back_populates="parent")
    test_cases: list["TestCase"] = Relationship(back_populates="suite")  # type: ignore
    
    __table_args__ = (
        Index("idx_suite_tenant_project", "tenant_id", "project_id"),
        Index("idx_suite_parent", "parent_id"),
        Index("idx_suite_path", "tenant_id", "path"),
    )

"""
Request/Approval models: TenantRequest, AccessRequest.

TenantRequest: User requests to create a new tenant (requires org admin approval)
AccessRequest: User requests access to an existing tenant (requires tenant admin approval)
"""

from uuid import UUID
from datetime import datetime

from sqlmodel import Field, SQLModel, Relationship, Column, Index
from sqlalchemy.dialects.postgresql import JSONB, TEXT

from .base import BaseModel, now_utc


# =============================================================================
# Database Models - TenantRequest
# =============================================================================

class TenantRequest(BaseModel, table=True):
    """
    TenantRequest table - user requests to create a new tenant.
    
    Workflow:
    1. Non-admin user submits tenant creation request
    2. Org admin reviews and approves/rejects
    3. On approval: tenant created, requester becomes admin
    """
    
    __tablename__ = "tenant_requests"
    
    # Requester
    requested_by: UUID = Field(foreign_key="users.id", index=True)
    
    # Requested tenant details
    tenant_name: str = Field(max_length=255)
    tenant_slug: str = Field(max_length=255, index=True)
    description: str | None = Field(
        default=None,
        sa_column=Column(TEXT, nullable=True),
    )
    justification: str = Field(
        sa_column=Column(TEXT, nullable=False),
    )
    
    # Proposed configuration
    config: dict = Field(
        default_factory=dict,
        sa_column=Column(JSONB, nullable=False, server_default="{}"),
    )
    
    # Status
    status: str = Field(
        max_length=50,
        default="pending_approval",
        index=True,
    )  # pending_approval, approved, rejected
    
    # Review
    reviewed_by: UUID | None = Field(
        default=None,
        foreign_key="users.id",
        nullable=True,
    )
    reviewed_at: datetime | None = None
    review_notes: str | None = Field(
        default=None,
        sa_column=Column(TEXT, nullable=True),
    )
    
    # Created tenant (if approved)
    tenant_id: UUID | None = Field(
        default=None,
        foreign_key="tenants.id",
        nullable=True,
    )
    
    # Relationships
    requester: "User" = Relationship(
        sa_relationship_kwargs={
            "foreign_keys": "[TenantRequest.requested_by]",
        }
    )  # type: ignore
    reviewer: "User | None" = Relationship(
        sa_relationship_kwargs={
            "foreign_keys": "[TenantRequest.reviewed_by]",
        }
    )  # type: ignore
    tenant: "Tenant | None" = Relationship()  # type: ignore
    
    __table_args__ = (
        Index("idx_tenant_request_status", "status"),
        Index("idx_tenant_request_requester", "requested_by", "status"),
    )


# =============================================================================
# Database Models - AccessRequest
# =============================================================================

class AccessRequest(BaseModel, table=True):
    """
    AccessRequest table - user requests access to a tenant.
    
    Workflow:
    1. User discovers tenant and requests access
    2. Tenant admin reviews and approves/rejects
    3. On approval: UserTenantRole created with requested role
    
    Note: Some tenants may have auto_approve_access_requests=true
    """
    
    __tablename__ = "access_requests"
    
    # Request details
    tenant_id: UUID = Field(foreign_key="tenants.id", index=True)
    requested_by: UUID = Field(foreign_key="users.id", index=True)
    requested_role: str = Field(max_length=50)  # 'admin', 'developer', 'viewer'
    
    # Justification
    reason: str = Field(
        sa_column=Column(TEXT, nullable=False),
    )
    
    # Status
    status: str = Field(
        max_length=50,
        default="pending_approval",
        index=True,
    )  # pending_approval, approved, rejected, auto_approved
    
    # Review
    reviewed_by: UUID | None = Field(
        default=None,
        foreign_key="users.id",
        nullable=True,
    )
    reviewed_at: datetime | None = None
    review_notes: str | None = Field(
        default=None,
        sa_column=Column(TEXT, nullable=True),
    )
    
    # Created role (if approved)
    user_tenant_role_id: UUID | None = Field(
        default=None,
        foreign_key="user_tenant_roles.id",
        nullable=True,
    )
    
    # Auto-approval
    auto_approved: bool = Field(default=False)
    
    # Relationships
    tenant: "Tenant" = Relationship()  # type: ignore
    requester: "User" = Relationship(
        sa_relationship_kwargs={
            "foreign_keys": "[AccessRequest.requested_by]",
        }
    )  # type: ignore
    reviewer: "User | None" = Relationship(
        sa_relationship_kwargs={
            "foreign_keys": "[AccessRequest.reviewed_by]",
        }
    )  # type: ignore
    role: "UserTenantRole | None" = Relationship()  # type: ignore
    
    __table_args__ = (
        Index("idx_access_request_tenant", "tenant_id", "status"),
        Index("idx_access_request_user", "requested_by", "tenant_id"),
    )

"""
System models: APIToken, AuditLog, SystemEvent, TestCoverage, TestFailureAnalysis.

APIToken: Service/user API tokens with scopes
AuditLog: User actions audit trail
SystemEvent: System-level events for monitoring
TestCoverage: Code coverage metrics
TestFailureAnalysis: Automated failure pattern analysis
"""

from uuid import UUID
from datetime import datetime

from sqlmodel import Field, SQLModel, Relationship, Column, Index

from .base import TenantBaseModel, BaseModel, now_utc

class APIToken(TenantBaseModel, table=True):
    """
    APIToken table - service and user API tokens.
    
    Supports:
    - Personal access tokens
    - Service tokens (Jenkins, CI/CD)
    - Fine-grained scopes
    """
    
    __tablename__ = "api_tokens"
    
    # Ownership
    user_id: UUID | None = Field(
        default=None,
        foreign_key="users.id",
        nullable=True,
        index=True,
    )
    
    # Identity
    name: str = Field(max_length=255)
    description: str | None = Field(
        default=None,
        sa_column=Column(TEXT, nullable=True),
    )
    
    # Token
    token_prefix: str = Field(max_length=20)  # First 8 chars, e.g., "qat_abcd"
    token_hash: str = Field(max_length=255)  # Hashed full token
    
    # Scopes
    scopes: list[str] = Field(
        default_factory=list,
        sa_column=Column(ARRAY(TEXT), nullable=False, server_default="{}"),
    )
    
    # Expiry
    expires_at: datetime | None = None
    
    # Status
    is_active: bool = Field(default=True)
    last_used_at: datetime | None = None
    
    # Metadata
    meta_data: dict = Field(
        default_factory=dict,
        sa_column=Column(JSONB, nullable=False, server_default="{}"),
    )
    
    # Relationships
    user: "User | None" = Relationship()  # type: ignore
    
    __table_args__ = (
        Index("idx_token_tenant_user", "tenant_id", "user_id"),
        Index("idx_token_prefix", "token_prefix"),
        Index("idx_token_active", "tenant_id", "is_active"),
    )

class SystemEvent(BaseModel, table=True):
    """
    SystemEvent table - system-level events for monitoring.
    
    NOTE: Not tenant-scoped - global events.
    
    Events:
    - Worker lifecycle (started, stopped, crashed)
    - Job execution (started, completed, failed)
    - System health (high memory, disk full)
    - Performance anomalies
    """
    
    __tablename__ = "system_events"
    
    # Event
    event_type: str = Field(max_length=100, index=True)
    severity: str = Field(max_length=20, default="info", index=True)  # info, warning, error, critical
    
    # Source
    source: str = Field(max_length=255, index=True)  # 'worker-01', 'scheduler', 'api'
    source_type: str = Field(max_length=50)  # 'worker', 'service', 'api'
    
    # Content
    message: str = Field(sa_column=Column(TEXT, nullable=False))
    details: dict = Field(
        default_factory=dict,
        sa_column=Column(JSONB, nullable=False, server_default="{}"),
    )
    
    # Context
    trace_id: str | None = Field(default=None, max_length=100)
    tags: list[str] = Field(
        default_factory=list,
        sa_column=Column(ARRAY(TEXT), nullable=False, server_default="{}"),
    )
    
    # Optional tenant context (for tenant-specific events)
    tenant_id: UUID | None = Field(default=None, index=True)
    
    __table_args__ = (
        Index("idx_system_event_type", "event_type", "severity"),
        Index("idx_system_event_source", "source", "source_type"),
        Index("idx_system_event_tenant", "tenant_id", "created_at"),
        Index("idx_system_event_created", "created_at"),
    )

class Tenant(BaseModel, table=True):
    """
    Tenant table - represents an organization or application workspace.
    
    Supports hierarchical structure: root organization → app tenants.
    """
    
    __tablename__ = "tenants"
    
    # Hierarchy
    parent_id: UUID | None = Field(
        default=None,
        foreign_key="tenants.id",
        nullable=True,
    )
    
    # Type and identification
    type: str = Field(max_length=50, default=TENANT_TYPE_APPLICATION)
    name: str = Field(max_length=255)
    slug: str = Field(max_length=255, unique=True, index=True)
    
    # Configuration (JSONB for flexibility)
    config: dict = Field(
        default_factory=dict,
        sa_column=Column(JSONB, nullable=False, server_default="{}"),
    )
    
    # Status
    status: str = Field(max_length=50, default=TENANT_STATUS_ACTIVE)
    
    # Quotas
    max_users: int | None = Field(default=100)
    max_storage_gb: int | None = Field(default=50)
    
    # Relationships
    parent: "Tenant | None" = Relationship(
        back_populates="children",
        sa_relationship_kwargs={
            "remote_side": "Tenant.id",
            "foreign_keys": "[Tenant.parent_id]",
        },
    )
    children: list["Tenant"] = Relationship(back_populates="parent")
    
    __table_args__ = (
        Index("idx_tenant_slug", "slug"),
        Index("idx_tenant_parent", "parent_id"),
    )


class UserTenantRole(BaseModel, table=True):
    """
    UserTenantRole table - maps users to tenants with specific roles.
    
    Enables multi-tenant access control:
    - One user can belong to multiple tenants
    - Each user-tenant pair has a role (admin, engineer, viewer, etc.)
    """
    
    __tablename__ = "user_tenant_roles"
    
    # Foreign keys
    user_id: UUID = Field(foreign_key="users.id", index=True)
    tenant_id: UUID = Field(foreign_key="tenants.id", index=True)
    
    # Role
    role: str = Field(max_length=50, default=ROLE_VIEWER)
    
    # Audit trail
    granted_at: datetime = Field(default_factory=lambda: datetime.utcnow())
    granted_by: UUID | None = Field(default=None, foreign_key="users.id")
    revoked_at: datetime | None = None
    
    # Relationships
    user: "User" = Relationship(back_populates="tenant_roles")  # type: ignore
    tenant: Tenant = Relationship()
    
    __table_args__ = (
        # Unique constraint: one role per user-tenant pair
        Index(
            "idx_unique_user_tenant",
            "user_id",
            "tenant_id",
            unique=True,
            postgresql_where="revoked_at IS NULL",  # Only active roles
        ),
        Index("idx_user_tenant", "user_id", "tenant_id"),
    )

class User(BaseModel, table=True):
    """
    User table - system users with authentication.
    
    Supports:
    - Local authentication (email + hashed_password)
    - SSO/LDAP (sso_provider + sso_id)
    """
    
    __tablename__ = "users"
    
    # Identity
    email: str = Field(max_length=255, unique=True, index=True)
    username: str = Field(max_length=100, unique=True, index=True)
    
    # Authentication
    hashed_password: str | None = Field(
        default=None,
        max_length=255,
    )  # None if SSO-only
    sso_provider: str | None = Field(default=None, max_length=50)  # 'ldap', 'okta', 'azure_ad'
    sso_id: str | None = Field(default=None, max_length=255)  # External ID from SSO
    
    # Profile
    full_name: str = Field(max_length=255)
    avatar_url: str | None = Field(default=None, max_length=500)
    
    # Status
    is_active: bool = Field(default=True)
    is_superuser: bool = Field(default=False)  # Global system admin
    
    # Timestamps
    last_login_at: datetime | None = None
    
    # Relationships
    tenant_roles: list["UserTenantRole"] = Relationship(back_populates="user")  # type: ignore
    
    __table_args__ = (
        Index("idx_user_email", "email"),
        Index("idx_user_username", "username"),
        Index("idx_user_sso", "sso_provider", "sso_id"),
    )

    TestWorker table - workers that execute tests.
    
    Types:
    - celery: Celery workers on VMs
    - jenkins: Jenkins agents
    - custom: Custom execution agents
    """
    
    __tablename__ = "test_workers"
    
    # Identity

    worker_type: str = Field(max_length=50, default=WORKER_TYPE_CELERY, index=True)

        sa_column=Column(JSONB, nullable=False, server_default="{}"),

    current_active_runs: int = Field(default=0)

    health_metrics: dict = Field(
        default_factory=dict,
        sa_column=Column(JSONB, nullable=False, server_default="{}"),
    )

        default=None,
        foreign_key="worker_templates.id",
        nullable=True,
    )
    
    # Metadata
    meta_data: dict = Field(
        default_factory=dict,
        sa_column=Column(JSONB, nullable=False, server_default="{}"),
    )
    
    # Relationships
    test_runs: list["TestRun"] = Relationship(back_populates="worker")  # type: ignore
    template: "WorkerTemplate | None" = Relationship(back_populates="workers")
